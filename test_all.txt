const fs = require('fs');
const path = require('path');
const { JSDOM } = require('jsdom');

const html = fs.readFileSync(path.join(__dirname, 'index.html'), 'utf-8');

// ===== 共通ヘルパー =====
async function setup(store) {
  const url = store === null || store === undefined
    ? 'https://example.test/'
    : `https://example.test/?store=${store}`;
  const dom = new JSDOM(html, { url, runScripts: 'dangerously', resources: 'usable' });
  await new Promise(res => dom.window.addEventListener('load', res));
  return dom;
}
function cat(dom, v) { dom.window.selCat(dom.window.document.querySelector(`.cat-btn[data-v="${v}"]`)); }
function rank(dom, v) { dom.window.selRank(dom.window.document.querySelector(`.btn.rank-${v}, .btn-rare.rank-${v}`)); }
function setCarrier(dom, value) {
  const { document } = dom.window;
  document.querySelectorAll('input[name=carrier]').forEach(r => { r.checked = false; });
  document.querySelector(`input[name=carrier][value="${value}"]`).checked = true;
  dom.window.onCarrierChange();
}
function makeMemo(dom, setupFn) {
  const { document } = dom.window;
  setupFn(dom);
  dom.window.generate();
  return document.getElementById('out-internal').textContent;
}

let PASS = 0, FAIL = 0;
function ok(cond, label) {
  if (cond) { PASS++; console.log(`  OK: ${label}`); }
  else { FAIL++; console.error(`  NG: ${label}`); }
}
function section(title) { console.log(`\n===== ${title} =====`); }

    for (const store of [null, 'osaka']) {
      const dom = await setup(store);
      const { document } = dom.window;
      let storeOk = true;

      for (const c of CATS) {
        for (const r of RANKS) {
          cat(dom, c);
          rank(dom, r);
          if (c === 'cymbal') document.getElementById('cymbal-diameter-input').value = '14';
          document.getElementById('size-input').value = '140';
          setCarrier(dom, 'ゆうパック');

          try {
            dom.window.generate();
          } catch (e) {
            console.error(`    例外: ${c}/${r}: ${e.message}`); storeOk = false; continue;
          }
          const out = document.getElementById('out-yahoo').textContent;
          if (!out || out.includes('⚠')) { console.error(`    出力不正: ${c}/${r}`); storeOk = false; continue; }
          if (/undefined|NaN/.test(out)) { console.error(`    undefined/NaN混入: ${c}/${r}`); storeOk = false; }

          // 動作の排他グループで二重チェックが起きていないか

          const isExclusiveGroup = checkedBase.length > 0 && checkedBase[0].hasAttribute('onchange');
          if (isExclusiveGroup && checkedBase.length > 1) {
            console.error(`    排他グループ二重チェック: ${c}/${r}: ${checkedBase.map(x=>x.value)}`); storeOk = false;
          }
        }

  section('6. 配送方法切替時の出力クリア（生成中断時の残留防止）');

    // JITBOXで生成→ゆうパックに切替→サイズ未入力で生成→前回出力が残らない

    ok(!out2.includes('JITBOX') && out2.includes('⚠'), 'サイズ未入力で生成失敗時、前回のJITBOX出力が残らずエラー表示される');

    document.getElementById('size-input').value = '140';
    dom.window.generate();
    const out3 = document.getElementById('out-yahoo').textContent;
    ok(!out3.includes('JITBOX') && out3.includes('ゆうパック・140サイズ'), 'サイズ入力後は正しくゆうパックの内容で生成される');

  {

      ['ゆうパック', 'ゆうパック・140サイズ'],
      ['佐川急便', '佐川急便・140サイズ'],
      ['ヤマト運輸', 'ヤマト運輸・140サイズ'],

      ['直接引取のみ', '当方倉庫での直接引取のみ'],

      setCarrier(dom, carrier);
      document.getElementById('size-input').value = '140';
      dom.window.generate();

      if (!out.includes(expect)) allOk = false;

    document.getElementById('jitbox-w').value = '400';

    const out = document.getElementById('out-yahoo').textContent;

    ok(document.getElementById('out-yahoo').textContent.includes('⚠'), 'シンバル直径0cmは生成が中断される');

  // ========================================================================
  section('10. 動作項目（排他選択グループ）の貼付復元バグ修正');
  // ========================================================================
  {
    // Cランク・オーディオで「通電確認済です。」のみ選択→復元しても二重チェックにならない
    const dom = await setup('ishikawa');
    const memo = makeMemo(dom, d => {
      cat(d, 'audio');
      const { document } = d.window;
      document.querySelectorAll('.op-base-chk').forEach(c => c.checked = false);

    dom2.window.generate();
    const out = document.getElementById('out-yahoo').textContent;
    ok(checked.length === 1 && checked[0] === '通電確認済です。' && !out.includes('出音確認済です。'),
      'Cランク: 貼付復元で「出音確認済」がデフォルト残留せず正しく1項目のみ復元される（現場報告の再現ケース）');

    const dom = await setup('ishikawa');
    const memo = makeMemo(dom, d => {
